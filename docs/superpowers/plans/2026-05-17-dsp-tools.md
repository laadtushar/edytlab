## Task 2: Parametric EQ tool

**Files:**
- Create: `crates/tools/src/tool/eq.rs`
- Modify: `crates/tools/src/tool/mod.rs`
- Modify: `crates/tools/src/dispatcher.rs`

A parametric EQ applies one or more biquad filter stages to all channels of the clip audio. Each band is described by `{ freq_hz, gain_db, q }`. We implement the Audio-EQ-Cookbook peak/shelf biquad directly in Rust (no new crate needed).

**Biquad coefficients (peak EQ band):**
```
A  = 10^(gain_db/40)
w0 = 2π * freq_hz / sample_rate
α  = sin(w0)/(2*Q)

b0 =  1 + α*A
b1 = -2*cos(w0)
b2 =  1 - α*A
a0 =  1 + α/A
a1 = -2*cos(w0)
a2 =  1 - α/A

Normalise by a0: b0/=a0, b1/=a0, b2/=a0, a1/=a0, a2/=a0
```

**Direct form II transposed (per sample, per channel):**
```
y[n] = b0*x[n] + w1
w1   = b1*x[n] - a1*y[n] + w2
w2   = b2*x[n] - a2*y[n]
```

//! `eq` — apply a parametric equalizer to a track clip.
//!
//! Each band is a peak-EQ biquad (Audio-EQ-Cookbook). Bands are applied
//! in series; the clip audio is decoded, filtered, and written to a new
//! derived WAV. Clip source_path is updated to the derived file.

use std::f32::consts::PI;
use serde::Deserialize;
use serde_json::{json, Value};

use crate::schema::{anthropic_tool, object_schema_with};
use crate::tool::util::{append_state, check_track_index, load_head_state, write_derived_wav};
use crate::{Tool, ToolContext, ToolResult};

#[derive(Debug, Deserialize, Clone)]
struct Band {
    freq_hz: f32,
    gain_db: f32,
    #[serde(default = "default_q")]
    q: f32,
}

fn default_q() -> f32 { 1.0 }

#[derive(Debug, Deserialize)]
struct Args {
    track: usize,
    bands: Vec<Band>,
}

pub struct EqTool;

/// Apply a single peak-EQ biquad band to interleaved samples.
/// `channels` is the interleave stride; each channel has independent state.
pub(crate) fn apply_biquad_band(
    samples: &[f32],
    channels: usize,
    freq_hz: f32,
    gain_db: f32,
    q: f32,
    sample_rate: u32,
) -> Vec<f32> {
    if gain_db == 0.0 || channels == 0 {
        return samples.to_vec();
    }

    let a = 10.0_f32.powf(gain_db / 40.0);
    let w0 = 2.0 * PI * freq_hz / sample_rate as f32;
    let alpha = w0.sin() / (2.0 * q);
    let cos_w0 = w0.cos();

    let a0 = 1.0 + alpha / a;
    let b0 = (1.0 + alpha * a) / a0;
    let b1 = (-2.0 * cos_w0) / a0;
    let b2 = (1.0 - alpha * a) / a0;
    let a1 = (-2.0 * cos_w0) / a0;
    let a2 = (1.0 - alpha / a) / a0;

    let frames = samples.len() / channels;
    let mut out = samples.to_vec();

    for ch in 0..channels {
        let mut w1 = 0.0f32;
        let mut w2 = 0.0f32;
        for f in 0..frames {
            let x = samples[f * channels + ch];
            let y = b0 * x + w1;
            w1 = b1 * x - a1 * y + w2;
            w2 = b2 * x - a2 * y;
            out[f * channels + ch] = y;
        }
    }
    out
}

impl Tool for EqTool {
    fn name(&self) -> &'static str { "eq" }

    fn schema(&self) -> Value {
        anthropic_tool(
            "eq",
            "Apply a parametric equalizer to a track. Each band is a peak-EQ filter defined by frequency, gain in dB, and Q factor. Decodes the clip, applies filters in series, writes a new derived WAV, and appends a session node.",
            serde_json::json!({
                "type": "object",
                "required": ["track", "bands"],
                "properties": {
                    "track": { "type": "integer", "description": "0-based track index" },
                    "bands": {
                        "type": "array",
                        "minItems": 1,
                        "items": {
                            "type": "object",
                            "required": ["freq_hz", "gain_db"],
                            "properties": {
                                "freq_hz": { "type": "number", "description": "Center frequency in Hz (20–20000)" },
                                "gain_db": { "type": "number", "description": "Gain in dB, positive = boost, negative = cut" },
                                "q": { "type": "number", "description": "Q factor (bandwidth); default 1.0" }
                            }
                        }
                    }
                }
            }),
        )
    }

    fn invoke(&self, args: Value, ctx: &mut ToolContext) -> crate::Result<ToolResult> {
        let args: Args = match serde_json::from_value(args) {
            Ok(a) => a,
            Err(e) => return Ok(ToolResult::Error(format!("invalid arguments: {e}"))),
        };

        if args.bands.is_empty() {
            return Ok(ToolResult::Error("bands must not be empty".into()));
        }
        for (i, b) in args.bands.iter().enumerate() {
            if b.freq_hz <= 0.0 || !b.freq_hz.is_finite() {
                return Ok(ToolResult::Error(format!("band[{i}]: freq_hz must be > 0")));
            }
            if !b.gain_db.is_finite() {
                return Ok(ToolResult::Error(format!("band[{i}]: gain_db must be finite")));
            }
            if b.q <= 0.0 || !b.q.is_finite() {
                return Ok(ToolResult::Error(format!("band[{i}]: q must be > 0")));
            }
        }

        let mut state = match load_head_state(ctx) {
            Ok(s) => s,
            Err(msg) => return Ok(ToolResult::Error(msg)),
        };

        if let Err(msg) = check_track_index(&state.tracks, args.track) {
            return Ok(ToolResult::Error(msg));
        }

        let clip = match state.tracks[args.track].clips.first().cloned() {
            Some(c) => c,
            None => return Ok(ToolResult::Error("track has no clips".into())),
        };

        let decoded = match audio_decoder::decode_file(&clip.source_path) {
            Ok(d) => d,
            Err(e) => return Ok(ToolResult::Error(format!("decode failed: {e}"))),
        };

        let channels = decoded.channels as usize;
        if channels == 0 {
            return Ok(ToolResult::Error("decoded source has 0 channels".into()));
        }

        // Apply bands in series
        let mut samples = decoded.samples.clone();
        for band in &args.bands {
            samples = apply_biquad_band(&samples, channels, band.freq_hz, band.gain_db, band.q, decoded.sample_rate);
        }

        let band_desc: Vec<String> = args.bands.iter()
            .map(|b| format!("{:.0}Hz {:+.1}dB", b.freq_hz, b.gain_db))
            .collect();
        let summary_str = band_desc.join(", ");

        let new_path = match write_derived_wav(ctx, &samples, decoded.channels, decoded.sample_rate) {
            Ok(p) => p,
            Err(e) => return Ok(ToolResult::Error(format!("write failed: {e}"))),
        };

        state.tracks[args.track].clips[0].source_path = new_path;

        let new_id = match append_state(
            ctx,
            state,
            format!("eq track {} [{}]", args.track, summary_str),
        ) {
            Ok(id) => id,
            Err(msg) => return Ok(ToolResult::Error(msg)),
        };

        Ok(ToolResult::Ok(json!({
            "node_id": new_id.to_hex(),
            "bands_applied": args.bands.len(),
            "summary": format!("Applied EQ to track {} ({} bands: {}); new head {}", args.track, args.bands.len(), summary_str, new_id.to_hex()),
        })))
    }
}

## Task 3: Dynamic compressor tool

**Files:**
- Create: `crates/tools/src/tool/compressor.rs`
- Modify: `crates/tools/src/tool/mod.rs`

**Compression algorithm (per-sample, per-channel):**
```
1. Compute instantaneous level: level = abs(x)
2. Envelope follower (smoothed RMS):
   if level > envelope:
       envelope = attack_coeff * envelope + (1 - attack_coeff) * level
   else:
       envelope = release_coeff * envelope + (1 - release_coeff) * level
   where attack_coeff  = exp(-1 / (attack_ms  * 0.001 * sample_rate))
         release_coeff = exp(-1 / (release_ms * 0.001 * sample_rate))
3. Gain reduction (in dB):
   if envelope_db > threshold_db:
       gain_db = threshold_db + (envelope_db - threshold_db) / ratio - envelope_db
   else:
       gain_db = 0.0
4. Apply makeup gain + gain reduction: y = x * 10^((gain_db + makeup_db) / 20)
```

//! `compressor` — dynamic range compressor.
//!
//! Uses a per-sample envelope follower with independent attack and
//! release coefficients. All channels share the same gain reduction
//! (peak of all channels drives the envelope). Decoded audio is
//! compressed and written to a new derived WAV.

use serde::Deserialize;
use serde_json::{json, Value};

use crate::schema::anthropic_tool;
use crate::tool::util::{append_state, check_track_index, load_head_state, write_derived_wav};
use crate::{Tool, ToolContext, ToolResult};

#[derive(Debug, Deserialize)]
struct Args {
    track: usize,
    threshold_db: f32,
    ratio: f32,
    attack_ms: f32,
    release_ms: f32,
    #[serde(default)]
    makeup_db: f32,
}

pub struct CompressorTool;

/// Pure compression function — decoupled from Tool for testability.
pub(crate) fn compress_samples(
    samples: &[f32],
    channels: usize,
    threshold_db: f32,
    ratio: f32,
    attack_ms: f32,
    release_ms: f32,
    makeup_db: f32,
    sample_rate: u32,
) -> Vec<f32> {
    if channels == 0 || samples.is_empty() {
        return samples.to_vec();
    }

    let sr = sample_rate as f32;
    let attack_coeff  = (-1.0 / (attack_ms  * 0.001 * sr)).exp();
    let release_coeff = (-1.0 / (release_ms * 0.001 * sr)).exp();
    let makeup_linear = 10.0_f32.powf(makeup_db / 20.0);

    let frames = samples.len() / channels;
    let mut out = vec![0.0f32; samples.len()];
    let mut envelope = 0.0f32;

    for f in 0..frames {
        // Peak across all channels for this frame
        let frame_peak = (0..channels)
            .map(|ch| samples[f * channels + ch].abs())
            .fold(0.0f32, f32::max);

        // Envelope follower
        let coeff = if frame_peak > envelope { attack_coeff } else { release_coeff };
        envelope = coeff * envelope + (1.0 - coeff) * frame_peak;

        // Gain reduction
        let envelope_db = if envelope > 1e-10 {
            20.0 * envelope.log10()
        } else {
            -200.0
        };
        let gain_reduction_db = if envelope_db > threshold_db {
            threshold_db + (envelope_db - threshold_db) / ratio - envelope_db
        } else {
            0.0
        };
        let gain_linear = 10.0_f32.powf(gain_reduction_db / 20.0) * makeup_linear;

        for ch in 0..channels {
            out[f * channels + ch] = (samples[f * channels + ch] * gain_linear).clamp(-1.0, 1.0);
        }
    }
    out
}

                "properties": {
                    "track": { "type": "integer" },
                    "threshold_db": { "type": "number", "description": "Compression threshold in dBFS (e.g. -12)" },
                    "ratio": { "type": "number", "description": "Compression ratio (e.g. 4 means 4:1)" },
                    "attack_ms": { "type": "number", "description": "Attack time in milliseconds (e.g. 5)" },
                    "release_ms": { "type": "number", "description": "Release time in milliseconds (e.g. 100)" },
                    "makeup_db": { "type": "number", "description": "Makeup gain in dB applied after compression (default 0)" }
                }
            }),
        )
    }

    fn invoke(&self, args: Value, ctx: &mut ToolContext) -> crate::Result<ToolResult> {
        let args: Args = match serde_json::from_value(args) {
            Ok(a) => a,
            Err(e) => return Ok(ToolResult::Error(format!("invalid arguments: {e}"))),
        };

        if args.ratio < 1.0 {
            return Ok(ToolResult::Error("ratio must be >= 1.0".into()));
        }
        if args.attack_ms <= 0.0 || !args.attack_ms.is_finite() {
            return Ok(ToolResult::Error("attack_ms must be > 0".into()));
        }
        if args.release_ms <= 0.0 || !args.release_ms.is_finite() {
            return Ok(ToolResult::Error("release_ms must be > 0".into()));
        }
        if !args.threshold_db.is_finite() {

        }

        let mut state = match load_head_state(ctx) {

        };

        if let Err(msg) = check_track_index(&state.tracks, args.track) {

        }

        let decoded = match audio_decoder::decode_file(&clip.source_path) {

        let new_id = match append_state(
            ctx,
            state,
            format!(
                "compress track {} {}:1 threshold {} dB",
                args.track, args.ratio, args.threshold_db

        ) {
            Ok(id) => id,
            Err(msg) => return Ok(ToolResult::Error(msg)),

        Ok(ToolResult::Ok(json!({
            "node_id": new_id.to_hex(),

                "Compressed track {} (threshold: {} dBFS, ratio: {}:1, attack: {}ms, release: {}ms, makeup: {:+}dB); new head {}",
                args.track, args.threshold_db, args.ratio, args.attack_ms, args.release_ms, args.makeup_db, new_id.to_hex()
            ),
        })))
    }
}

**Algorithm (spectral subtraction, per-frame):**
```
1. Profile phase: average magnitude spectrum over first N frames (noise-only region)
2. Subtraction phase: for each STFT frame:
   a. FFT the windowed frame (Hann window)
   b. noise_estimate = profile_magnitude * strength  (strength in [0,1])
   c. subtracted_magnitude = max(frame_magnitude - noise_estimate, 0) * (1 - floor_ratio) + floor_ratio * frame_magnitude
   d. IFFT with original phase
3. Overlap-add reconstruction (50% overlap)
```

**`realfft` usage:**
```rust

let fft = planner.plan_fft_forward(fft_size);
let ifft = planner.plan_fft_inverse(fft_size);
```

//! `noise_reduction` — spectral subtraction noise reduction.
//!
//! Requires a `noise_duration_sec` specifying how many seconds at the
//! start of the clip are "noise only" (the profile region). The spectral
//! profile is the average magnitude per bin over that window. The
//! remaining audio has the estimated noise spectrum subtracted from each
//! STFT frame using overlap-add reconstruction.
//!
//! Uses `realfft` (already a workspace dep via audio-analysis).

use serde::Deserialize;
use serde_json::{json, Value};
use realfft::RealFftPlanner;

use crate::schema::anthropic_tool;
use crate::tool::util::{append_state, check_track_index, load_head_state, write_derived_wav};
use crate::{Tool, ToolContext, ToolResult};

const FFT_SIZE: usize = 2048;
const HOP_SIZE: usize = 512; // 75% overlap

/// Compute mean magnitude spectrum over the first `profile_duration_sec` of `samples`.
pub(crate) fn compute_noise_profile(
    samples: &[f32],
    channels: usize,
    fft_size: usize,
    profile_duration_sec: f32,
) -> Vec<f32> {
    // Use channel 0 only for profile
    let profile_frames = (profile_duration_sec * 44100.0) as usize; // approx; actual sr unused in tests
    let num_bins = fft_size / 2 + 1;
    let mut profile = vec![0.0f32; num_bins];
    let mut count = 0usize;

    let mut planner = RealFftPlanner::<f32>::new();
    let fft = planner.plan_fft_forward(fft_size);

    // Hann window
    let window: Vec<f32> = (0..fft_size)
        .map(|i| 0.5 * (1.0 - (2.0 * std::f32::consts::PI * i as f32 / fft_size as f32).cos()))
        .collect();

    let mono_samples: Vec<f32> = if channels > 1 {
        samples.chunks(channels).map(|c| c[0]).collect()
    } else {
        samples.to_vec()
    };

    let hop = fft_size / 4;
    let mut pos = 0;
    while pos + fft_size <= mono_samples.len().min(profile_frames * channels.max(1)) {
        let mut frame: Vec<f32> = mono_samples[pos..pos + fft_size]
            .iter().enumerate()
            .map(|(i, &s)| s * window[i])
            .collect();
        let mut spectrum = fft.make_output_vec();
        fft.process(&mut frame, &mut spectrum).ok();
        for (b, c) in spectrum.iter().enumerate() {
            profile[b] += (c.re * c.re + c.im * c.im).sqrt();
        }
        count += 1;
        pos += hop;
    }

    if count > 0 {
        for v in &mut profile { *v /= count as f32; }
    }
    profile
}

#[derive(Debug, Deserialize)]
struct Args {
    track: usize,
    noise_duration_sec: f32,
    #[serde(default = "default_strength")]
    strength: f32,
    #[serde(default = "default_floor")]
    floor: f32,
}
fn default_strength() -> f32 { 0.85 }
fn default_floor() -> f32 { 0.05 }

pub struct NoiseReductionTool;

/// Reduce noise in-place using spectral subtraction on a mono channel.
/// `samples` is mono (single channel). Modifies in place.
pub(crate) fn reduce_noise(
    samples: &mut Vec<f32>,
    _channels: usize, // reserved; we process mono extracted separately
    _noise_duration_sec: f32,
    strength: f32,
    floor: f32,
    sample_rate: u32,
) {
    let fft_size = FFT_SIZE;
    let hop = HOP_SIZE;
    let num_bins = fft_size / 2 + 1;

    let profile_samples = (0.1 * sample_rate as f32) as usize; // first 100ms as profile
    let window: Vec<f32> = (0..fft_size)
        .map(|i| 0.5 * (1.0 - (2.0 * std::f32::consts::PI * i as f32 / fft_size as f32).cos()))
        .collect();

    let mut planner = RealFftPlanner::<f32>::new();
    let fft = planner.plan_fft_forward(fft_size);
    let ifft = planner.plan_fft_inverse(fft_size);

    // Build noise profile from first 100ms
    let mut profile = vec![0.0f32; num_bins];
    let mut profile_count = 0usize;
    {
        let mut pos = 0;
        while pos + fft_size <= profile_samples.min(samples.len()) {
            let mut frame: Vec<f32> = samples[pos..pos + fft_size]
                .iter().enumerate()
                .map(|(i, &s)| s * window[i])
                .collect();
            let mut spectrum = fft.make_output_vec();
            if fft.process(&mut frame, &mut spectrum).is_ok() {
                for (b, c) in spectrum.iter().enumerate() {
                    profile[b] += (c.re * c.re + c.im * c.im).sqrt();
                }
                profile_count += 1;
            }
            pos += hop;
        }
        if profile_count > 0 {
            for v in &mut profile { *v /= profile_count as f32; }
        }
    }

    // Overlap-add output buffer
    let out_len = samples.len();
    let mut out = vec![0.0f32; out_len];
    let mut norm = vec![0.0f32; out_len];

    let mut pos = 0;
    while pos + fft_size <= out_len {
        let mut frame: Vec<f32> = samples[pos..pos + fft_size]
            .iter().enumerate()
            .map(|(i, &s)| s * window[i])
            .collect();
        let mut spectrum = fft.make_output_vec();
        if fft.process(&mut frame, &mut spectrum).is_err() {
            pos += hop;
            continue;
        }

        // Spectral subtraction
        for (b, c) in spectrum.iter_mut().enumerate() {
            let mag = (c.re * c.re + c.im * c.im).sqrt();
            let phase = c.im.atan2(c.re);
            let noise_est = profile[b] * strength;
            let new_mag = (mag - noise_est).max(mag * floor);
            c.re = new_mag * phase.cos();
            c.im = new_mag * phase.sin();
        }

        let mut time_out = ifft.make_output_vec();
        if ifft.process(&mut spectrum, &mut time_out).is_ok() {
            let scale = 1.0 / fft_size as f32;
            for (i, &s) in time_out.iter().enumerate() {
                let idx = pos + i;
                if idx < out_len {
                    out[idx] += s * scale * window[i];
                    norm[idx] += window[i] * window[i];
                }
            }
        }
        pos += hop;
    }

    // Normalize overlap-add
    for (i, s) in samples.iter_mut().enumerate() {
        if norm[i] > 1e-10 {
            *s = out[i] / norm[i];
        }
    }
}

impl Tool for NoiseReductionTool {
    fn name(&self) -> &'static str { "noise_reduction" }

    fn schema(&self) -> Value {
        anthropic_tool(
            "noise_reduction",
            "Reduce background noise from a track using spectral subtraction. Profiles the noise from the first noise_duration_sec seconds, then subtracts the estimated noise spectrum from the rest of the audio. Decodes the clip, processes in-place, writes a derived WAV, and appends a session node.",
            serde_json::json!({
                "type": "object",
                "required": ["track", "noise_duration_sec"],
                "properties": {
                    "track": { "type": "integer" },
                    "noise_duration_sec": {
                        "type": "number",
                        "description": "Seconds at the start of the clip that contain only background noise (used to build the noise profile)"
                    },
                    "strength": {
                        "type": "number",
                        "description": "How aggressively to subtract noise (0.0–1.0, default 0.85)"
                    },
                    "floor": {
                        "type": "number",
                        "description": "Minimum fraction of original magnitude to retain (prevents musical noise artifacts, default 0.05)"
                    }
                }
            }),
        )
    }

    fn invoke(&self, args: Value, ctx: &mut ToolContext) -> crate::Result<ToolResult> {
        let args: Args = match serde_json::from_value(args) {
            Ok(a) => a,
            Err(e) => return Ok(ToolResult::Error(format!("invalid arguments: {e}"))),
        };

        if args.noise_duration_sec <= 0.0 || !args.noise_duration_sec.is_finite() {
            return Ok(ToolResult::Error("noise_duration_sec must be > 0".into()));
        }
        if args.strength < 0.0 || args.strength > 1.0 {
            return Ok(ToolResult::Error("strength must be in [0.0, 1.0]".into()));
        }
        if args.floor < 0.0 || args.floor > 1.0 {
            return Ok(ToolResult::Error("floor must be in [0.0, 1.0]".into()));
        }

        let mut state = match load_head_state(ctx) {
            Ok(s) => s,
            Err(msg) => return Ok(ToolResult::Error(msg)),
        };

        if let Err(msg) = check_track_index(&state.tracks, args.track) {
            return Ok(ToolResult::Error(msg));
        }

        let clip = match state.tracks[args.track].clips.first().cloned() {
            Some(c) => c,
            None => return Ok(ToolResult::Error("track has no clips".into())),
        };

        let decoded = match audio_decoder::decode_file(&clip.source_path) {
            Ok(d) => d,
            Err(e) => return Ok(ToolResult::Error(format!("decode failed: {e}"))),
        };

        let channels = decoded.channels as usize;
        if channels == 0 {
            return Ok(ToolResult::Error("decoded source has 0 channels".into()));
        }

        // Process each channel independently
        let frames = decoded.samples.len() / channels;
        let mut result = decoded.samples.clone();

        for ch in 0..channels {
            let mut mono: Vec<f32> = (0..frames)
                .map(|f| decoded.samples[f * channels + ch])
                .collect();
            reduce_noise(&mut mono, 1, args.noise_duration_sec, args.strength, args.floor, decoded.sample_rate);
            for f in 0..frames {
                result[f * channels + ch] = mono[f];
            }
        }

        let new_path = match write_derived_wav(ctx, &result, decoded.channels, decoded.sample_rate) {
            Ok(p) => p,
            Err(e) => return Ok(ToolResult::Error(format!("write failed: {e}"))),
        };

        state.tracks[args.track].clips[0].source_path = new_path;

        let new_id = match append_state(
            ctx,
            state,
            format!(
                "noise_reduction track {} profile={:.2}s strength={:.0}%",
                args.track, args.noise_duration_sec, args.strength * 100.0
            ),
        ) {
            Ok(id) => id,
            Err(msg) => return Ok(ToolResult::Error(msg)),
        };

        Ok(ToolResult::Ok(json!({
            "node_id": new_id.to_hex(),
            "summary": format!(
                "Applied noise reduction to track {} (profile: {:.2}s, strength: {:.0}%); new head {}",
                args.track, args.noise_duration_sec, args.strength * 100.0, new_id.to_hex()
            ),
        })))
    }
}
