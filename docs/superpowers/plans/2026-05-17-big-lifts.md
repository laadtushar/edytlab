> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver four high-impact features requiring significant cross-layer changes: stem separation UX (model download + post-separation timeline refresh), word-level transcript sync (click-to-seek in transcript panel), agent memory UI (view/edit memory in settings), and a basic EQ+compressor effect chain.

**Architecture:** Stem separation UX wraps the existing `separate_stems` tool with a download flow for the Demucs ONNX model and automatic timeline refresh after separation. Transcript sync adds a `get_transcript` Tauri command and a `TranscriptPanel` component that highlights the current word during playback. Agent memory UI exposes the existing `read_memory`/`write_memory` commands in a Settings tab. Effect chain removes the `EffectsUnsupportedInPhase1` guard and implements biquad EQ and soft-knee compressor in the render pipeline, plus three new tools.

**Tech Stack:** React 19, TypeScript, Tauri 2, Rust (`audio-engine`, `session`, `tools` crates), Tauri HTTP plugin (model download), biquad DSP (pure Rust, no extra deps).

```rust
#[tauri::command]
pub async fn download_demucs_model(
    app_handle: tauri::AppHandle,
    window: tauri::Window,
) -> Result<String, CommandError> {
    use std::io::Write;

    let data_dir = app_handle
        .path()
        .app_data_dir()
        .map_err(|_| CommandError::Io(std::io::Error::other("app data dir unavailable")))?;
    let models_dir = data_dir.join("models");
    std::fs::create_dir_all(&models_dir).map_err(CommandError::Io)?;
    let model_path = models_dir.join("demucs-4stems.onnx");

    // Replace MODEL_URL with the actual CDN URL for the ONNX export.
    // This must be a stable, publicly accessible URL to the Demucs 4-stem ONNX model.
    const MODEL_URL: &str = "https://dl.edytlab.app/models/demucs-4stems-v4.onnx";

    let resp = reqwest::get(MODEL_URL)
        .await
        .map_err(|e| CommandError::Io(std::io::Error::other(e.to_string())))?;
    let bytes = resp
        .bytes()
        .await
        .map_err(|e| CommandError::Io(std::io::Error::other(e.to_string())))?;
    std::fs::write(&model_path, &bytes).map_err(CommandError::Io)?;

    Ok(model_path.to_string_lossy().into_owned())
}
```

```rust
#[derive(Debug, serde::Serialize)]
pub struct TranscriptWordDto {
    pub text: String,
    pub start_s: f32,
    pub end_s: f32,
    pub confidence: f32,
}

#[tauri::command]
pub async fn get_transcript(
    state: tauri::State<'_, AppState>,
) -> Result<Vec<TranscriptWordDto>, CommandError> {
    let store = lock_std(&state.store, "store")?;
    let head = match store.head() {
        Some(h) => h,
        None => return Ok(vec![]),
    };
    let node = store.get(head).map_err(CommandError::Session)?;
    let words = node
        .state
        .transcript
        .map(|t| {
            t.words
                .into_iter()
                .map(|w| TranscriptWordDto {
                    text: w.text,
                    start_s: w.start_s,
                    end_s: w.end_s,
                    confidence: w.confidence,
                })
                .collect()
        })
        .unwrap_or_default();
    Ok(words)
}
```

//! DSP effects: biquad EQ and soft-knee compressor.

use serde_json::Value;
use session::EffectInstance;

// ---------------------------------------------------------------------------
// Biquad filter (Direct Form II Transposed)
// ---------------------------------------------------------------------------

pub struct BiquadFilter {
    b0: f32, b1: f32, b2: f32,
    a1: f32, a2: f32,
    s1: f32, s2: f32,
}

impl BiquadFilter {
    /// Parametric (peaking) EQ band.
    /// `freq`: center frequency Hz, `q`: bandwidth factor, `gain_db`: boost/cut in dB.
    pub fn parametric_eq(freq: f32, sample_rate: f32, q: f32, gain_db: f32) -> Self {
        let a = 10.0_f32.powf(gain_db / 40.0);
        let w0 = 2.0 * std::f32::consts::PI * freq / sample_rate;
        let alpha = w0.sin() / (2.0 * q);
        let cos_w0 = w0.cos();
        let b0 = 1.0 + alpha * a;
        let b1 = -2.0 * cos_w0;
        let b2 = 1.0 - alpha * a;
        let a0 = 1.0 + alpha / a;
        let a1 = -2.0 * cos_w0;
        let a2 = 1.0 - alpha / a;
        Self {
            b0: b0 / a0, b1: b1 / a0, b2: b2 / a0,
            a1: a1 / a0, a2: a2 / a0,
            s1: 0.0, s2: 0.0,
        }
    }

    pub fn process(&mut self, x: f32) -> f32 {
        let y = self.b0 * x + self.s1;
        self.s1 = self.b1 * x - self.a1 * y + self.s2;
        self.s2 = self.b2 * x - self.a2 * y;
        y
    }
}

// ---------------------------------------------------------------------------
// Soft-knee compressor
// ---------------------------------------------------------------------------

pub struct CompressorParams {
    pub threshold_db: f32,
    pub ratio: f32,
    pub attack_ms: f32,
    pub release_ms: f32,
    pub makeup_db: f32,
    pub knee_db: f32,
}

pub struct Compressor {
    params: CompressorParams,
    envelope: f32,
    attack_coeff: f32,
    release_coeff: f32,
}

impl Compressor {
    pub fn new(params: CompressorParams, sample_rate: f32) -> Self {
        let attack_coeff = (-1.0 / (params.attack_ms * 0.001 * sample_rate)).exp();
        let release_coeff = (-1.0 / (params.release_ms * 0.001 * sample_rate)).exp();
        Self { params, envelope: 0.0, attack_coeff, release_coeff }
    }

    /// Returns gain reduction in dB (≤ 0) for a given input level in dB.
    pub fn compute_gain_reduction(&mut self, input_db: f32) -> f32 {
        let knee_half = self.params.knee_db / 2.0;
        let over = input_db - self.params.threshold_db;
        let cs = if over < -knee_half {
            0.0
        } else if over > knee_half {
            (1.0 / self.params.ratio - 1.0) * over
        } else {
            (1.0 / self.params.ratio - 1.0) * (over + knee_half).powi(2) / (2.0 * self.params.knee_db)
        };
        cs
    }

    pub fn process(&mut self, sample: f32) -> f32 {
        let abs = sample.abs();
        let coeff = if abs > self.envelope { self.attack_coeff } else { self.release_coeff };
        self.envelope = abs + coeff * (self.envelope - abs);
        let env_db = if self.envelope > 1e-10 {
            20.0 * self.envelope.log10()
        } else {
            -120.0
        };
        let gr = self.compute_gain_reduction(env_db);
        let makeup = 10.0_f32.powf(self.params.makeup_db / 20.0);
        sample * 10.0_f32.powf(gr / 20.0) * makeup
    }
}

// ---------------------------------------------------------------------------
// Effect chain
// ---------------------------------------------------------------------------

pub enum Effect {
    Eq { bands: Vec<BiquadFilter> },
    Compressor(Compressor),
}

impl Effect {
    pub fn process(&mut self, sample: f32) -> f32 {
        match self {
            Effect::Eq { bands } => bands.iter_mut().fold(sample, |s, b| b.process(s)),
            Effect::Compressor(c) => c.process(sample),
        }
    }
}

/// Parse an `EffectInstance` from the session schema into a runnable `Effect`.
/// Returns `None` for unknown or bypassed effects (bypassed = no-op).
pub fn build_effect(inst: &EffectInstance) -> Option<Effect> {
    if inst.bypassed { return None; }
    match inst.kind.as_str() {
        "eq" => {
            let bands_val = inst.params.get("bands")?;
            let bands_arr = bands_val.as_array()?;
            let mut filters = Vec::new();
            for b in bands_arr {
                let freq = b.get("freq_hz")?.as_f64()? as f32;
                let q = b.get("q").and_then(|v| v.as_f64()).unwrap_or(1.0) as f32;
                let gain_db = b.get("gain_db")?.as_f64()? as f32;
                let sr = b.get("sample_rate").and_then(|v| v.as_f64()).unwrap_or(44100.0) as f32;
                filters.push(BiquadFilter::parametric_eq(freq, sr, q, gain_db));
            }
            Some(Effect::Eq { bands: filters })
        }
        "compressor" => {
            let p = &inst.params;
            Some(Effect::Compressor(Compressor::new(
                CompressorParams {
                    threshold_db: p.get("threshold_db").and_then(|v| v.as_f64()).unwrap_or(-18.0) as f32,
                    ratio:        p.get("ratio").and_then(|v| v.as_f64()).unwrap_or(2.0) as f32,
                    attack_ms:    p.get("attack_ms").and_then(|v| v.as_f64()).unwrap_or(10.0) as f32,
                    release_ms:   p.get("release_ms").and_then(|v| v.as_f64()).unwrap_or(100.0) as f32,
                    makeup_db:    p.get("makeup_db").and_then(|v| v.as_f64()).unwrap_or(0.0) as f32,
                    knee_db:      p.get("knee_db").and_then(|v| v.as_f64()).unwrap_or(2.0) as f32,
                },
                44100.0,
            )))
        }
        _ => None,
    }
}

                    "track_index": { "type": "integer", "minimum": 0 },
                    "kind":        { "type": "string", "enum": ["eq", "compressor"] },
                    "params":      { "type": "object" },
                },
                "required": ["track_index", "kind", "params"],
                "additionalProperties": false,
            }),
        )
    }

    fn invoke(&self, args: Value, ctx: &mut ToolContext) -> crate::Result<ToolResult> {
        let args: Args = match serde_json::from_value(args) {
            Ok(a) => a,
            Err(e) => return Ok(ToolResult::Error(format!("invalid arguments: {e}"))),
        };
        let head = match ctx.store.head() {
            Some(h) => h,
            None => return Ok(ToolResult::Error("no session loaded".into())),
        };
        let mut node = ctx.store.get(head).map_err(|e| crate::DispatchError::Internal(e.to_string()))?;
        let track = match node.state.tracks.get_mut(args.track_index) {
            Some(t) => t,
            None => return Ok(ToolResult::Error(format!(
                "track_index {} out of range", args.track_index
            ))),
        };
        track.effects.push(EffectInstance {
            kind: args.kind.clone(),
            params: args.params,
            bypassed: false,
        });
        let new_node = SessionNode { parent: Some(head), state: node.state, ..Default::default() };
        let new_id = ctx.store.append(new_node).map_err(|e| crate::DispatchError::Internal(e.to_string()))?;
        Ok(ToolResult::Ok(json!({
            "new_node_id": new_id.to_hex(),
            "effect_index": track.effects.len() - 1,
            "summary": format!("Added {} effect to track {}", args.kind, args.track_index),
        })))
    }
}
